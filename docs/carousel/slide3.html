def retrieve_with_gate(
    vectorstore: Chroma, query: str, k: int = 4, threshold: float = 0.48
) -> tuple[list[Document], float, bool]:
    # 1. Measure cosine similarity across top-k chunks
    results = vectorstore.similarity_search_with_relevance_scores(query, k=k)
    max_sim = max([score for _, score in results]) if results else 0.0

    # 2. Intercept OOD queries at vector level (0 tokens billed)
    if max_sim < threshold:
        logger.info("Gate rejected query '%s' (sim=%.3f < %.2f)", query, max_sim, threshold)
        return [], max_sim, False  # < 25ms execution

    return [doc for doc, _ in results], max_sim, True
